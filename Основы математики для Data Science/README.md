### 5. Интерполяция и полиномы
Восстановление полинома по точкам (степень = кол-во точек − 1).

```python
a2, a1, a0 = symbols("a2 a1 a0")

eq1 = a2*(-2.5)**2 + a1*(-2.5) + a0 - 15
eq2 = a2*(-2)**2   + a1*(-2)   + a0 - 10
eq3 = a2*(-1.5)**2 + a1*(-1.5) + a0 - 7

nonlinsolve([eq1, eq2, eq3], [a2, a1, a0])
```

### 8. Аппроксимация
Подбор лучших коэффициентов по минимальному MSE.

```python
a2, a1, a0 = symbols('a2 a1 a0')
MSE = 1/3 * (((a2*2 + a1*200 + a0) - 200)**2 +
             ((a2*1 + a1*450 + a0) - 300)**2 +
             ((a2*3 + a1*550 + a0) - 600)**2)

MSE2 = diff(MSE, a2)
MSE1 = diff(MSE, a1)
MSE0 = diff(MSE, a0)
nonlinsolve([MSE2, MSE1, MSE0], [a2, a1, a0])
```

### 10. Градиент и градиентный спуск
Минимизация MSE пошаговым движением против градиента.

```python
a1, a2 = symbols('a1 a2')
MSE = 1/4 * ((a1*1 + a2*2 - 5)**2 +
             (a1*5 + a2*3 - 6)**2 +
             (a1*2 + a2*4 - 10)**2 +
             (a1*3 + a2*7 - 8)**2)

grad_MSE1 = diff(MSE, a1)
grad_MSE2 = diff(MSE, a2)

a1, a2 = 0.0, 0.0
step = 0.01

while True:
    grad1 = 19.5*a1 + 23.0*a2 - 39.5
    grad2 = 23.0*a1 + 39.0*a2 - 62.0
    a1 = a1 - step * grad1
    a2 = a2 - step * grad2
    current_MSE = 1/4 * ((a1*1 + a2*2 - 5)**2 + ...)
    if current_MSE <= 6.25:
        break
```
