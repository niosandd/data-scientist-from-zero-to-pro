# Записываем данные без изменения в Data Lake в бронзовый слой в папку с датой
import pyspark.sql.functions as F

ingestion_df = ingestion_df.withColumn("date", F.date_sub(F.current_date(), 1))

# Write the DataFrame to the specified path in append mode
ingestion_df.write.partitionBy("date").mode("append").text("s3://your_name_bucket/bronze/sensor_data/")

# Проверяем что записалось по количеству строк (приложить скриншот что assert прошел успешно)
assert spark.read.text("s3://your_name_bucket/bronze/sensor_data/date=2023-12-19").count() == ingestion_df.count()

# Создаем агрегаты
from pyspark.sql import functions as F

silver_agg_df = silver_df.groupBy("device_id", "name", "date").agg(F.round(F.avg("heartrate"), 2).alias("avg_heartrate"), F.round(F.stddev("heartrate"), 2).alias("stddev_heartrate"), F.round(F.min("heartrate"), 2).alias("min_heartrate"), F.round(F.max("heartrate"), 2).alias("max_heartrate"))

# Проверяем результат
silver_agg_df.show(5, truncate=False)
